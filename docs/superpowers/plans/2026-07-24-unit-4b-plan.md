# Unit 4B Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build "Unit 4B", a 10–15 minute PSX-style first-person apartment horror game (React Three Fiber) with a two-act structure: calm exploration → entity chase/escape.

**Architecture:** One persistent R3F `<Canvas>`; a zustand store holding a single `phase` flag is the backbone. Entity AI is a pure state machine gated by `phase`, navigating a hardcoded room-graph via BFS. Collision is custom capsule-vs-AABB (no physics engine). The PSX look comes from a low-res render + vertex-snap shader injection + dither/CRT post passes. All art is procedural (primitives + canvas textures).

**Tech Stack:** Vite, React 18 + TypeScript, @react-three/fiber v8, @react-three/drei v9, three, zustand, howler, @react-three/postprocessing v2, vitest.

## Global Constraints

- `phase: 'title' | 'calm' | 'chase' | 'win' | 'gameover'` — the single backbone flag, in the zustand store.
- Entity mounts/activates ONLY when `phase === 'chase'`.
- No physics engine, no navmesh, no new major dependencies without asking.
- Internal render resolution: `RESOLUTION_SCALE = 0.25` (≈480×270 at 1080p), nearest-neighbor upscale via CSS `image-rendering: pixelated`.
- All textures: `NearestFilter` min+mag, mipmaps off.
- Lighting: MeshLambertMaterial / vertex colors only. No PBR, no realtime shadows.
- Fog: `THREE.FogExp2`, dark, aggressive.
- Player: capsule r=0.3 h=1.7, walk 2.6 m/s, sprint 4.5 m/s, stamina 100 (drain 20/s, regen 10/s after 1s).
- Entity speeds: search 1.6 m/s, chase 3.4 m/s. Catch radius 1.2m.
- Hidden-door puzzle: exactly one step (shove box off rug).
- Checkpoint: snapshot taken when the hatch opens; retry restores it.
- Vitest covers pure logic only (store, collision, room graph BFS, AI machine, objective chain). Visuals are verified by manual playtest.
- Git: conventional commits, push after each commit, one `feat/` branch per milestone, PR + merge to main per milestone. Milestone branches: M1 `feat/m1-scaffold`, M2 `feat/m2-psx-pipeline`, M3 `feat/m3-apartment`, M4 `feat/m4-act1`, M5 `feat/m5-entity`, M6 `feat/m6-chase`, M7 `feat/m7-audio`, M8 `feat/m8-art`, M9 `feat/m9-ui-polish`.
- Shell: run all commands inside WSL (`wsl -e bash -lc '...'` from PowerShell, or plain bash inside WSL). Project root: `/home/jose/projects/horror-game`.

```ts
import { create } from 'zustand'

export type Phase = 'title' | 'calm' | 'chase' | 'win' | 'gameover'
export type EntityState = 'dormant' | 'alert' | 'search' | 'chase' | 'attack'

export interface Objective {
  id: string
  label: string
  complete: boolean
}

export interface GameFlags {
  leaseRead: boolean
  rugNoticed: boolean
  hatchOpen: boolean
  voicemailPlayed: boolean
}

export interface Checkpoint {
  playerPosition: [number, number, number]
  playerYaw: number
  objectives: Objective[]
  activeObjectiveId: string | null
  flags: GameFlags
}

export const INITIAL_OBJECTIVES: Objective[] = [
  { id: 'unpack', label: 'Unpack your boxes', complete: false },
  { id: 'lease', label: 'Look over the lease', complete: false },
  { id: 'rug', label: 'Check under the rug', complete: false },
  { id: 'hatch', label: "See what's down there", complete: false },
  { id: 'escape', label: 'GET OUT', complete: false },
]

const INITIAL_FLAGS: GameFlags = {
  leaseRead: false,
  rugNoticed: false,
  hatchOpen: false,
  voicemailPlayed: false,
}

interface GameState {
  phase: Phase
  objectives: Objective[]
  activeObjectiveId: string | null
  player: { position: [number, number, number]; stamina: number; hidden: boolean }
  entity: { state: EntityState; room: string | null; targetRoom: string | null }
  flags: GameFlags
  checkpoint: Checkpoint | null
  setPhase: (phase: Phase) => void
  completeObjective: (id: string) => void
  setActiveObjective: (id: string | null) => void
  setPlayerPosition: (p: [number, number, number]) => void
  setStamina: (v: number) => void
  setHidden: (hidden: boolean) => void
  setEntityState: (state: EntityState) => void
  setEntityRoom: (room: string | null, targetRoom?: string | null) => void
  setFlag: <K extends keyof GameFlags>(key: K, value: GameFlags[K]) => void
  saveCheckpoint: (playerYaw: number) => void
  restoreCheckpoint: () => void
  resetGame: () => void
}

function freshState() {
  return {
    phase: 'title' as Phase,
    objectives: INITIAL_OBJECTIVES.map((o) => ({ ...o })),
    activeObjectiveId: null as string | null,
    player: { position: [0, 0, 0] as [number, number, number], stamina: 100, hidden: false },
    entity: { state: 'dormant' as EntityState, room: null as string | null, targetRoom: null as string | null },
    flags: { ...INITIAL_FLAGS },
    checkpoint: null as Checkpoint | null,
  }
}

export const useGameStore = create<GameState>((set, get) => ({
  ...freshState(),
  setPhase: (phase) => set({ phase }),
  completeObjective: (id) =>
    set((s) => ({
      objectives: s.objectives.map((o) => (o.id === id ? { ...o, complete: true } : o)),
    })),
  setActiveObjective: (id) => set({ activeObjectiveId: id }),
  setPlayerPosition: (p) => set((s) => ({ player: { ...s.player, position: p } })),
  setStamina: (v) => set((s) => ({ player: { ...s.player, stamina: v } })),
  setHidden: (hidden) => set((s) => ({ player: { ...s.player, hidden } })),
  setEntityState: (state) => set((s) => ({ entity: { ...s.entity, state } })),
  setEntityRoom: (room, targetRoom = null) =>
    set((s) => ({ entity: { ...s.entity, room, targetRoom } })),
  setFlag: (key, value) => set((s) => ({ flags: { ...s.flags, [key]: value } })),
  saveCheckpoint: (playerYaw) => {
    const s = get()
    set({
      checkpoint: {
        playerPosition: [...s.player.position],
        playerYaw,
        objectives: s.objectives.map((o) => ({ ...o })),
        activeObjectiveId: s.activeObjectiveId,
        flags: { ...s.flags },
      },
    })
  },
  restoreCheckpoint: () => {
    const cp = get().checkpoint
    if (!cp) return
    set({
      phase: 'calm',
      player: { position: [...cp.playerPosition], stamina: 100, hidden: false },
      entity: { state: 'dormant', room: null, targetRoom: null },
      objectives: cp.objectives.map((o) => ({ ...o })),
      activeObjectiveId: cp.activeObjectiveId,
      flags: { ...cp.flags },
    })
  },
  resetGame: () => set(freshState()),
}))
```

`src/systems/player/collision.ts`:

```ts
export interface AABB {
  min: [number, number, number]
  max: [number, number, number]
}

function clamp(v: number, lo: number, hi: number) {
  return Math.max(lo, Math.min(hi, v))
}

/** True if a vertical capsule (feet at pos) overlaps the box. */
export function capsuleHitsAABB(
  pos: [number, number, number],
  radius: number,
  height: number,
  box: AABB,
): boolean {
  const segLo = pos[1] + radius
  const segHi = pos[1] + height - radius
  const cy = clamp(box.min[1], segLo, segHi)
  const px = clamp(pos[0], box.min[0], box.max[0])
  const py = clamp(cy, box.min[1], box.max[1])
  const pz = clamp(pos[2], box.min[2], box.max[2])
  const dx = pos[0] - px
  const dy = cy - py
  const dz = pos[2] - pz
  return dx * dx + dy * dy + dz * dz < radius * radius
}

/**
 * Move a capsule by velocity*dt, resolving against AABBs.
 * Returns corrected feet position. Horizontal resolution only (flat floors);
 * anti-tunnel: displacement is sub-stepped so no step exceeds radius/2.
 */
export function resolveCapsule(
  pos: [number, number, number],
  radius: number,
  height: number,
  boxes: AABB[],
  velocity: [number, number, number],
  dt: number,
): [number, number, number] {
  const maxStep = radius / 2
  const dist = Math.hypot(velocity[0], velocity[2]) * dt
  const steps = Math.max(1, Math.ceil(dist / maxStep))
  const sub: [number, number, number] = [
    (velocity[0] * dt) / steps,
    (velocity[1] * dt) / steps,
    (velocity[2] * dt) / steps,
  ]
  let p: [number, number, number] = [...pos]
  for (let s = 0; s < steps; s++) {
    p = [p[0] + sub[0], p[1] + sub[1], p[2] + sub[2]]
    for (let iter = 0; iter < 3; iter++) {
      for (const box of boxes) {
        if (!capsuleHitsAABB(p, radius, height, box)) continue
        // push out horizontally along smallest penetration axis
        const cx = (box.min[0] + box.max[0]) / 2
        const cz = (box.min[2] + box.max[2]) / 2
        const dxPos = p[0] - cx
        const dzPos = p[2] - cz
        const overlapX = (box.max[0] - box.min[0]) / 2 + radius - Math.abs(dxPos)
        const overlapZ = (box.max[2] - box.min[2]) / 2 + radius - Math.abs(dzPos)
        if (overlapX < overlapZ) {
          p[0] = cx + Math.sign(dxPos || 1) * ((box.max[0] - box.min[0]) / 2 + radius)
        } else {
          p[2] = cz + Math.sign(dzPos || 1) * ((box.max[2] - box.min[2]) / 2 + radius)
        }
      }
    }
  }
  return p
}
```

```tsx
  const hidden = useGameStore((s) => s.player.hidden)
  const setStamina = useGameStore((s) => s.setStamina)

    const down = (e: KeyboardEvent) => { keys.current[e.code] = true }
    const up = (e: KeyboardEvent) => { keys.current[e.code] = false }

    window.addEventListener('keyup', up)

    playerRig.position.set(...spawn)
    playerRig.yaw = spawnYaw
    camera.position.set(spawn[0], spawn[1] + PLAYER.eyeHeight, spawn[2])
    camera.rotation.set(0, spawnYaw, 0, 'YXZ')
  }, [spawn, spawnYaw, camera])

    const dt = Math.min(rawDt, 0.05)
    const canMove = locked.current && (phase === 'calm' || phase === 'chase') && !hidden
    const k = keys.current
    const f = (canMove && KEYS.forward.some((c) => k[c]) ? 1 : 0) - (canMove && KEYS.back.some((c) => k[c]) ? 1 : 0)
    const r = (canMove && KEYS.right.some((c) => k[c]) ? 1 : 0) - (canMove && KEYS.left.some((c) => k[c]) ? 1 : 0)

    const speed = wantSprint ? PLAYER.sprintSpeed : PLAYER.walkSpeed

    // stamina
    if (wantSprint && (f !== 0 || r !== 0)) {
      stamina.current = Math.max(0, stamina.current - PLAYER.staminaDrainPerSec * dt)
      regenWait.current = PLAYER.staminaRegenDelaySec
    } else {
      regenWait.current -= dt
      if (regenWait.current <= 0) stamina.current = Math.min(PLAYER.staminaMax, stamina.current + PLAYER.staminaRegenPerSec * dt)
    }

    // movement in camera-yaw space
    const yaw = camera.rotation.y
    const dir = new THREE.Vector3(-Math.sin(yaw) * f + Math.cos(yaw) * r, 0, -Math.cos(yaw) * f - Math.sin(yaw) * r)
    if (dir.lengthSq() > 0) dir.normalize().multiplyScalar(speed)
    const next = resolveCapsule(
      [playerRig.position.x, playerRig.position.y, playerRig.position.z],
      PLAYER.radius, PLAYER.height, walls, [dir.x, 0, dir.z], dt,
    )
    playerRig.position.set(...next)
    playerRig.yaw = yaw
    playerRig.moving = dir.lengthSq() > 0
    playerRig.sprinting = wantSprint && playerRig.moving
    camera.position.set(next[0], next[1] + PLAYER.eyeHeight, next[2])

    // sync discrete state to store at ~10Hz (stamina bar, AI reads rig directly)
    lastStoreSync.current += dt
    if (lastStoreSync.current > 0.1) {
      lastStoreSync.current = 0
      setStamina(Math.round(stamina.current))
      useGameStore.getState().setPlayerPosition([next[0], next[1], next[2]])
    }
  })

  return (
    <PointerLockControls
      onLock={() => { locked.current = true }}
      onUnlock={() => { locked.current = false }}
    />
  )
}
```

```ts
import * as THREE from 'three'
import { INTERNAL_RESOLUTION } from '../game/constants'

/**
 * MeshLambertMaterial with PS1 vertex-snap: gl_Position is quantized to the
 * low-res pixel grid, producing authentic geometry wobble.
 * Update uniforms.uResolution at runtime if the drawing buffer size changes.
 */
export function makePSXMaterial(
  params: THREE.MeshLambertMaterialParameters = {},
): THREE.MeshLambertMaterial {
  const mat = new THREE.MeshLambertMaterial(params)
  mat.onBeforeCompile = (shader) => {
    shader.uniforms.uResolution = {
      value: new THREE.Vector2(INTERNAL_RESOLUTION.width, INTERNAL_RESOLUTION.height),
    }
    shader.vertexShader = shader.vertexShader
      .replace('#include <common>', '#include <common>\nuniform vec2 uResolution;')
      .replace(
        '#include <project_vertex>',
        `#include <project_vertex>
        {
          vec2 grid = uResolution * 0.5;
          gl_Position.xy = floor(gl_Position.xy / gl_Position.w * grid + 0.5) / grid * gl_Position.w;
        }`,
      )
  }
  // distinct program cache key per material so the injection compiles once
  mat.customProgramCacheKey = () => 'psx-lambert'
  return mat
}
```

```ts
import * as THREE from 'three'
import { INTERNAL_RESOLUTION } from '../../game/constants'

/** Bayer 4x4 ordered dithering + color quantization. */
export const DitherShader = {
  uniforms: {
    tDiffuse: { value: null as THREE.Texture | null },
    uResolution: { value: new THREE.Vector2(INTERNAL_RESOLUTION.width, INTERNAL_RESOLUTION.height) },
    uLevels: { value: 32.0 },
  },
  vertexShader: /* glsl */ `
    varying vec2 vUv;
    void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }
  `,
  fragmentShader: /* glsl */ `
    uniform sampler2D tDiffuse;
    uniform vec2 uResolution;
    uniform float uLevels;
    varying vec2 vUv;

    float bayer(vec2 p) {
      int x = int(mod(p.x, 4.0));
      int y = int(mod(p.y, 4.0));
      int i = y * 4 + x;
      // 4x4 bayer matrix / 16
      float m[16];
      m[0]=0.0;  m[1]=8.0;  m[2]=2.0;  m[3]=10.0;
      m[4]=12.0; m[5]=4.0;  m[6]=14.0; m[7]=6.0;
      m[8]=3.0;  m[9]=11.0; m[10]=1.0; m[11]=9.0;
      m[12]=15.0;m[13]=7.0; m[14]=13.0;m[15]=5.0;
      for (int k = 0; k < 16; k++) { if (k == i) return m[k] / 16.0; }
      return 0.0;
    }

    void main() {
      vec2 frag = vUv * uResolution;
      vec3 c = texture2D(tDiffuse, vUv).rgb;
      float d = bayer(floor(frag)) - 0.5;
      c = floor(c * uLevels + d + 0.5) / uLevels;
      gl_FragColor = vec4(c, 1.0);
    }
  `,
}
```

```ts
/** Subtle CRT: faint scanlines + slight chromatic aberration toward edges. */
export const CrtShader = {
  uniforms: {
    tDiffuse: { value: null as THREE.Texture | null },
    uResolution: { value: new THREE.Vector2(INTERNAL_RESOLUTION.width, INTERNAL_RESOLUTION.height) },
    uScanline: { value: 0.08 },   // 0 = off
    uAberration: { value: 0.0015 },
  },
  vertexShader: /* glsl */ `
    varying vec2 vUv;
    void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }
  `,
  fragmentShader: /* glsl */ `
    uniform sampler2D tDiffuse;
    uniform vec2 uResolution;
    uniform float uScanline;
    uniform float uAberration;
    varying vec2 vUv;

    void main() {
      vec2 centered = vUv - 0.5;
      float edge = dot(centered, centered);
      vec2 off = centered * edge * uAberration * 40.0;
      float r = texture2D(tDiffuse, vUv + off).r;
      float g = texture2D(tDiffuse, vUv).g;
      float b = texture2D(tDiffuse, vUv - off).b;
      vec3 c = vec3(r, g, b);
      float line = step(0.5, fract(vUv.y * uResolution.y * 0.5));
      c *= 1.0 - uScanline * line;
      gl_FragColor = vec4(c, 1.0);
    }
  `,
}
```

```ts
    bounds: { min: [x0, y0, z0], max: [x1, y1, z1] },
  }

export const ROOMS: Record<RoomId, RoomNode> = {
  living: room('living', -2.5, 2.0, 0.5, 4.5),
  entry: room('entry', 2.0, 8.5, 2.5, 4.5),
  hallway: room('hallway', 2.0, 3.5, -4.5, 0.5),

  bedroom: room('bedroom', -0.5, 2.0, -4.5, -1.0),
  bathroom: room('bathroom', -2.5, -0.5, -4.5, -2.0),
  secret: room('secret', -2.0, 1.0, 1.0, 4.0, -4, -0.2),
}

export const ADJACENCY: Record<RoomId, RoomId[]> = {
  living: ['entry', 'hallway'],
  entry: ['living'],
  hallway: ['living', 'kitchen', 'bedroom'],
  kitchen: ['hallway'],
  bedroom: ['hallway', 'bathroom'],
  bathroom: ['bedroom'],
  secret: [],
}

export const DOORWAYS: Record<string, [number, number, number]> = {
  'living|entry': [2.0, 0, 3.5],
  'hallway|living': [2.75, 0, 0.5],
  'hallway|kitchen': [3.5, 0, -3.0],
  'bedroom|hallway': [2.0, 0, -2.75],
  'bathroom|bedroom': [-0.5, 0, -3.25],
}

export function findRoomForPoint(p: [number, number, number]): RoomId | null {
  for (const r of Object.values(ROOMS)) {
    const yOk = r.id === 'secret' ? p[1] < -0.5 : p[1] >= -0.5
    if (
      yOk &&
      p[0] >= r.bounds.min[0] && p[0] <= r.bounds.max[0] &&
      p[2] >= r.bounds.min[2] && p[2] <= r.bounds.max[2]
    ) return r.id
  }
  return null
}

export function findPath(from: RoomId, to: RoomId): RoomId[] {
  if (from === to) return [from]
  const prev = new Map<RoomId, RoomId | null>([[from, null]])
  const queue: RoomId[] = [from]
  while (queue.length) {
    const cur = queue.shift()!
    for (const next of ADJACENCY[cur]) {
      if (prev.has(next)) continue
      prev.set(next, cur)
      if (next === to) {
        const path: RoomId[] = [to]
        let p: RoomId | null = cur
        while (p) { path.unshift(p); p = prev.get(p) ?? null }
        return path
      }
      queue.push(next)
    }
  }
  return [from] // unreachable (should not happen for above-ground rooms)
}

export function doorwayBetween(a: RoomId, b: RoomId): [number, number, number] {
  const d = DOORWAYS[`${a}|${b}`] ?? DOORWAYS[`${b}|${a}`]
  if (!d) throw new Error(`no doorway between ${a} and ${b}`)
  return d
}
```

```ts
import type * as THREE from 'three'

export interface Interactable {
  id: string
  object: THREE.Object3D
  label: () => string
  enabled: () => boolean
  onInteract: () => void
  /** If set, E must be held this long to trigger (progress shown in UI). */

  registry.set(i.id, i)

}

export function getInteractables(): Interactable[] {
  return [...registry.values()]

export function _clearRegistry(): void {
  registry.clear()
}
```

```ts
export interface TimelineStep {
  at: number // seconds since start
  do: () => void
}
export interface TimelineState {
  elapsed: number
  fired: Set<number>
}
export function advanceTimeline(state: TimelineState, steps: TimelineStep[], elapsed: number): TimelineState {
  const fired = new Set(state.fired)
  steps.forEach((s, i) => {
    if (elapsed >= s.at && !fired.has(i)) {
      fired.add(i)
      s.do()
    }
  })
  return { elapsed, fired }
}
```

```ts
import { ENTITY } from '../../game/constants'
import type { RoomId } from './roomGraph'

export type AIState = 'dormant' | 'alert' | 'search' | 'chase' | 'attack'

export interface AIContext {
  state: AIState
  lastKnownPlayerRoom: RoomId | null
  targetRoom: RoomId | null
  alertTimer: number
  sawPlayerHide: boolean
}

export function initialAIContext(): AIContext {
  return { state: 'dormant', lastKnownPlayerRoom: null, targetRoom: null, alertTimer: 0, sawPlayerHide: false }
}

export type AIEvent =
  | { type: 'WAKE' }
  | {
      type: 'TICK'
      dt: number
      playerVisible: boolean
      playerRoom: RoomId | null
      playerHidden: boolean
      distanceToPlayer: number
    }

export function aiReducer(ctx: AIContext, ev: AIEvent): AIContext {
  if (ctx.state === 'attack') return ctx
  if (ev.type === 'WAKE') {
    return ctx.state === 'dormant'
      ? { ...ctx, state: 'alert', alertTimer: ENTITY.alertDurationSec }
      : ctx
  }
  const { dt, playerVisible, playerRoom, playerHidden, distanceToPlayer } = ev
  switch (ctx.state) {
    case 'dormant':
      return ctx
    case 'alert': {
      const alertTimer = ctx.alertTimer - dt
      const lastKnownPlayerRoom = playerRoom ?? ctx.lastKnownPlayerRoom
      return alertTimer <= 0
        ? { ...ctx, state: 'chase', alertTimer: 0, lastKnownPlayerRoom, targetRoom: lastKnownPlayerRoom }
        : { ...ctx, alertTimer, lastKnownPlayerRoom }
    }
    case 'chase': {
      // caught: close enough, and either sees you or watched you hide
      if (distanceToPlayer <= ENTITY.catchRadius && (playerVisible || ctx.sawPlayerHide || !playerHidden)) {
        return { ...ctx, state: 'attack' }
      }
      if (playerVisible && playerRoom) {
        return { ...ctx, lastKnownPlayerRoom: playerRoom, targetRoom: playerRoom, sawPlayerHide: playerHidden }
      }
      // lost the player
      return { ...ctx, state: 'search', targetRoom: ctx.lastKnownPlayerRoom }
    }
    case 'search': {
      if (playerVisible && !playerHidden && playerRoom) {
        return { ...ctx, state: 'chase', lastKnownPlayerRoom: playerRoom, targetRoom: playerRoom }
      }
      return ctx
    }
    default:
      return ctx
  }
}
```

```ts
import type { AABB } from '../player/collision'

type V3 = [number, number, number]

/** Slab-method segment vs AABB. */
export function segmentIntersectsAABB(a: V3, b: V3, box: AABB): boolean {
  let tmin = 0
  let tmax = 1
  for (let i = 0; i < 3; i++) {
    const d = b[i] - a[i]
    if (Math.abs(d) < 1e-9) {
      if (a[i] < box.min[i] || a[i] > box.max[i]) return false
    } else {
      let t1 = (box.min[i] - a[i]) / d
      let t2 = (box.max[i] - a[i]) / d
      if (t1 > t2) [t1, t2] = [t2, t1]
      tmin = Math.max(tmin, t1)
      tmax = Math.min(tmax, t2)
      if (tmin > tmax) return false
    }
  }
  return true
}

export function hasLineOfSight(from: V3, to: V3, walls: AABB[]): boolean {
  for (const w of walls) {
    if (segmentIntersectsAABB(from, to, w)) return false
  }
  return true
}

export function canHear(entityPos: V3, playerPos: V3, playerSprinting: boolean, hearingRadius: number): boolean {
  if (!playerSprinting) return false
  const dx = entityPos[0] - playerPos[0]
  const dz = entityPos[2] - playerPos[2]
  return dx * dx + dz * dz <= hearingRadius * hearingRadius
}
```
